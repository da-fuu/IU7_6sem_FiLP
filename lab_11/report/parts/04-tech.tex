\section{Реализация программы}

Реализация программы представлена в листинге \ref{lst:prog}.

\begin{lstlisting}[caption={Реализация программы}, label={lst:prog}]
list_length(List, Res) :-
    rec_len(List, 0, Res).

rec_len([], Len, Len) :- !.
rec_len([_|Rest], Len, Res) :-
    NewLen is Len + 1,
    rec_len(Rest, NewLen, Res).


list_sum(List, Res) :-
    rec_sum(List, 0, Res).

rec_sum([], Sum, Sum) :- !.
rec_sum([X|Rest], Sum, Res) :-
    NewSum is Sum + X,
    rec_sum(Rest, NewSum, Res).


list_odd_sum(List, Res) :-
    rec_odd_sum(List, 0, Res).

rec_odd_sum([], Sum, Sum) :- !.
rec_odd_sum([_], Sum, Sum) :- !.
rec_odd_sum([_, X | Rest], Sum, Res) :-
    NewSum is Sum + X,
    rec_odd_sum(Rest, NewSum, Res).


list_gt([], _, []) :- !.
list_gt([X | Rest], Num, [X | ResRest]) :-
    X > Num,
    !,
    list_gt(Rest, Num, ResRest).
list_gt([_ | Rest], Num, ResList) :-
    list_gt(Rest, Num, ResList).


list_del_one([], _, []) :- !.
list_del_one([Elem | Rest], Elem, Rest) :- !.
list_del_one([X | Rest], Elem, [X | ResRest]) :-
    list_del_one(Rest, Elem, ResRest).


list_del_many([], _, []) :- !.
list_del_many([Elem | Rest], Elem, ResList) :-
    !,
    list_del_many(Rest, Elem, ResList).
list_del_many([X | Rest], Elem, [X | ResRest]) :-
    list_del_many(Rest, Elem, ResRest).


list_join([], BList, BList) :- !.
list_join([X | ARest], BList, [X | ResRest]) :-
    list_join(ARest, BList, ResRest).

\end{lstlisting}

\section{Варианты вопросов и ответов на них}

\begin{lstlisting}[caption={Вопрос и ответ №1}, label={lst:q1}]
?- list_length([1, 2, 3], Res).
Res = 3.
\end{lstlisting}

\begin{lstlisting}[caption={Вопрос и ответ №2}, label={lst:q2}]
?- list_sum([1, 2, 3], Res).
Res = 6.
\end{lstlisting}

\begin{lstlisting}[caption={Вопрос и ответ №3}, label={lst:q3}]
?- list_odd_sum([1, 2, 3], Res).
Res = 2.
\end{lstlisting}

\begin{lstlisting}[caption={Вопрос и ответ №4}, label={lst:q4}]
?- list_gt([1, 2, 3, 1, 3], 2, Res).
Res = [3, 3].
\end{lstlisting}

\begin{lstlisting}[caption={Вопрос и ответ №5}, label={lst:q5}]
?- list_del_one([1, 2, 3, 1, 3], 3, Res).
Res = [1, 2, 1, 3].
\end{lstlisting}

\begin{lstlisting}[caption={Вопрос и ответ №6}, label={lst:q6}]
?- list_del_many([1, 2, 3, 1, 3], 3, Res).
Res = [1, 2, 1].
\end{lstlisting}

\begin{lstlisting}[caption={Вопрос и ответ №7}, label={lst:q7}]
?- list_join([1, 2, 3, 1, 3], [3, 2, 1], Res).
Res = [1, 2, 3, 1, 3, 3, 2, 1].
\end{lstlisting}
